```typescript
// FRONT-01: Estrutura Atomic Design existe em src/components/
// STATUS: RED — implementado em 07-02-PLAN.md
import { describe, it, expect } from "vitest";
import { existsSync, readdirSync } from "node:fs";
import { resolve, dirname } from "node:path";
import { fileURLToPath } from "node:url";

const __dirname = dirname(fileURLToPath(import.meta.url));
const srcComponents = resolve(__dirname, "../../components");

function hasAtLeastOneTsx(dir: string): boolean {
  if (!existsSync(dir)) return false;
  return readdirSync(dir).some((f) => f.endsWith(".tsx"));
}

describe("FRONT-01: Atomic Design — estrutura de diretórios", () => {
  it("src/components/atoms/ existe com pelo menos 1 arquivo .tsx", () => {
    expect(hasAtLeastOneTsx(`${srcComponents}/atoms`)).toBe(true);
  });

  it("src/components/molecules/ existe com pelo menos 1 arquivo .tsx", () => {
    expect(hasAtLeastOneTsx(`${srcComponents}/molecules`)).toBe(true);
  });

  it("src/components/organisms/ existe com pelo menos 1 arquivo .tsx", () => {
    expect(hasAtLeastOneTsx(`${srcComponents}/organisms`)).toBe(true);
  });

  it("src/components/templates/ existe com pelo menos 1 arquivo .tsx", () => {
    expect(hasAtLeastOneTsx(`${srcComponents}/templates`)).toBe(true);
  });

  it("src/components/pages/ existe com pelo menos 1 arquivo .tsx", () => {
    expect(hasAtLeastOneTsx(`${srcComponents}/pages`)).toBe(true);
  });
});
```
