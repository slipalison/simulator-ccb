### Task 4: GREEN getProfileClient tests
**File:** `frontend/src/tests/profile-components.test.tsx` (update)

Implement the API client test stubs:

```typescript
import { getProfileClient, ProfileError } from '@/lib/api';

describe('getProfileClient', () => {
  beforeEach(() => {
    vi.resetModules();
  });

  it('fetches profile with Bearer token', async () => {
    const mockToken = 'fake-jwt-token';
    const mockProfile = {
      id: 'test-id',
      name: 'Test User',
      email: 'test@email.com',
      phone: '(11) 99999-9999',
      type: 'PessoaFisica',
      cpf: '123.456.789-00',
    };

    // Mock fetch
    global.fetch = vi.fn().mockResolvedValue({
      ok: true,
      json: () => Promise.resolve(mockProfile),
    });

    // Mock auth context
    vi.doMock('@/lib/auth-context', () => ({
      getAccessToken: () => mockToken,
    }));

    const result = await getProfileClient();

    expect(result).toEqual(mockProfile);
    expect(global.fetch).toHaveBeenCalledWith('/api/clients/me', {
      method: 'GET',
      headers: {
        'Authorization': `Bearer ${mockToken}`,
        'Content-Type': 'application/json',
      },
    });
  });

  it('throws ProfileError when no token available', async () => {
    vi.doMock('@/lib/auth-context', () => ({
      getAccessToken: () => null,
    }));

    await expect(getProfileClient()).rejects.toThrow(ProfileError);
  });

  it('throws ProfileError on 401 response', async () => {
    global.fetch = vi.fn().mockResolvedValue({
      ok: false,
      status: 401,
    });

    vi.doMock('@/lib/auth-context', () => ({
      getAccessToken: () => 'expired-token',
    }));

    await expect(getProfileClient()).rejects.toThrow(ProfileError);
  });
});
```
