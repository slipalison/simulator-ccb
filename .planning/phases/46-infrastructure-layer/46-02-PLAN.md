  ```csharp
  public async Task AddAsync(Cedente cedente, CancellationToken ct = default)
  {
      // Set shadow properties from CedenteDocumento DU per D-09
      var entry = _db.Entry(cedente);
      cedente.Documento.Match(
          pf => { entry.Property("DocumentoTipo").CurrentValue = "PF"; entry.Property<string?>("CpfValue").CurrentValue = pf.Cpf.Value; entry.Property<string?>("CnpjCedenteValue").CurrentValue = null; },
          pj => { entry.Property("DocumentoTipo").CurrentValue = "PJ"; entry.Property<string?>("CpfValue").CurrentValue = null; entry.Property<string?>("CnpjCedenteValue").CurrentValue = pj.Cnpj.Value; }
      );
      await _db.Cedentes.AddAsync(cedente, ct);
      await _db.SaveChangesAsync(ct);
  }
  ```

  ```csharp
  public async Task<bool> ExistsByDocumentoAsync(CedenteDocumento documento, Guid companyId, CancellationToken ct = default)
  {
      return documento.Match(
          pf => _db.Cedentes.IgnoreQueryFilters()
              .AnyAsync(c => c.ClienteId == companyId
                  && EF.Property<string>(c, "DocumentoTipo") == "PF"
                  && EF.Property<string>(c, "CpfValue") == pf.Cpf.Value, ct),
          pj => _db.Cedentes.IgnoreQueryFilters()
              .AnyAsync(c => c.ClienteId == companyId
                  && EF.Property<string>(c, "DocumentoTipo") == "PJ"
                  && EF.Property<string>(c, "CnpjCedenteValue") == pj.Cnpj.Value, ct)
      );
  }
  ```

| Boundary | Description |
|----------|-------------|
| Repository → AppDbContext | Repositories use IgnoreQueryFilters for uniqueness checks — prevents HasQueryFilter from hiding cross-tenant duplicates |
| AppDbContext → PostgreSQL | Migration enforces DB-level constraints as defense-in-depth |
| Global TipoAtivo | No tenant filter — any authenticated user can see all tipos de ativo (by design per TEN-03) |
| CedenteDocumento DU | Shadow properties queried with both type and value — PF/PJ confusion not possible per D-09/D-10 |

| T-46-07 | Tampering | CedenteRepository ExistsByDocumento | mitigate | Shadow property queries check DocumentoTipo + value — PF/PJ confusion not possible per D-09 |
| T-46-08 | Spoofing | Migration | mitigate | Unique constraints at DB level as final safety net for application-level dedup logic |
