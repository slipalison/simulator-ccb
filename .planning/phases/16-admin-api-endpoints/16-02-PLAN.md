```csharp
public sealed class AdminRepository : IAdminRepository
{
    private readonly AppDbContext _db;

    public async Task<Client?> GetByIdAsync(Guid id, CancellationToken ct = default)
        => await _db.Clients.FindAsync([id], ct);

    public async Task<(IReadOnlyList<Client> Items, int TotalCount)> GetPagedAsync(
        int page, int pageSize, string? search, string? status, CancellationToken ct = default)
    {
        var query = _db.Clients.AsNoTracking();

        // Status filter: deleted
        if (status?.ToLowerInvariant() == "deleted")
            query = query.Where(c => c.DeletedAt.HasValue);
        else
            query = query.Where(c => !c.DeletedAt.HasValue);

        // Search filter
        if (!string.IsNullOrWhiteSpace(search))
            query = ApplySearch(query, search);

        var totalCount = await query.CountAsync(ct);

        var items = await query
            .OrderBy(c => c.Name)
            .ThenBy(c => c.Id)
            .Skip((page - 1) * pageSize)
            .Take(pageSize)
            .ToListAsync(ct);

        return (items.AsReadOnly(), totalCount);
    }

    public async Task UpdateAsync(Client client, CancellationToken ct = default)
    {
        _db.Clients.Update(client);
        await Task.CompletedTask;
    }

    public async Task SaveChangesAsync(CancellationToken ct = default)
        => await _db.SaveChangesAsync(ct);

    private static IQueryable<Client> ApplySearch(IQueryable<Client> query, string search)
    {
        var normalized = search.Trim().ToLowerInvariant();
        var digitsOnly = new string(normalized.Where(char.IsDigit).ToArray());

        return query.Where(c =>
            EF.Functions.ILike(c.Name, $"%{normalized}%") ||
            EF.Functions.ILike(c.Email.Value, $"%{normalized}%") ||
            (c.Cpf != null && c.Cpf.Value.Contains(normalized)) ||
            (c.Cnpj != null && c.Cnpj.Value.Contains(normalized)) ||
            (digitsOnly.Length > 0 && (
                (c.Cpf != null && c.Cpf.Value.Contains(digitsOnly)) ||
                (c.Cnpj != null && c.Cnpj.Value.Contains(digitsOnly))
            ))
        );
    }
}
```

### Task 5: BlockUserCommandHandler (ADMIN-04)

**Type:** Application

**Implementation:**
1. `IAdminRepository.GetByIdAsync(command.UserId)` — throw if null
2. Get Keycloak user: `IKeycloakUserService.GetUserByEmailAsync(client.Email.Value)`

**Idempotency:** If user is already disabled, the operation is a no-op (still write audit log but skip Keycloak call).

**Verification:**

- Keycloak unavailable: throws `InvalidOperationException` with generic message

### Task 7: DeleteUserCommandHandler (ADMIN-05) — LGPD

**Type:** Application
**Scope:** `Onboarding.Application/Admin/Commands/DeleteUserCommand.cs`

**Implementation (order matters — compensation strategy):**
1. `IAdminRepository.GetByIdAsync(command.UserId)` — throw if null
2. **Validate confirm email:** If `command.ConfirmEmail` doesn't match `client.Email.Value`, throw `ArgumentException("Email confirmation does not match.")`
3. **Check already deleted:** If `client.IsDeleted`, throw `InvalidOperationException("User is already deleted.")`
4. **Capture snapshot BEFORE anonymization:** Serialize all PII fields for audit
5. **Anonymize:** `client.Anonymize()` + `IAdminRepository.UpdateAsync(client)` + `SaveChangesAsync()`
6. **Delete from Keycloak:**
   ```csharp
   try
   {
       await _keycloakUserService.DeleteUserByEmailAsync(client.Email.Value, ct);
       // Wait — email is now "deleted-{id}@internal.local". Use the ORIGINAL email captured before anonymization.
   }
   catch (Exception ex)
   {
       _logger.LogCritical(ex, "Keycloak deletion failed for user {Email} after DB anonymization", originalEmail);
       // Set pending flag or just log — PII is already scrubbed
   }
   ```
7. **Audit log:** `AuditActions.UserDeleted` with `SnapshotBefore` = full PII, `SnapshotAfter` = anonymized state

**CRITICAL:** Capture the original email BEFORE calling `client.Anonymize()`. After anonymization, the email is `deleted-{id}@internal.local` which won't match any Keycloak user.

**Compensation:** If Keycloak delete fails, log as CRITICAL and continue. Do NOT rollback anonymization — PII is already gone and user expects deletion.

**Verification:**
- Client anonymized in app_db
- Keycloak user deleted
- Audit log created with full PII snapshot before anonymization
- Keycloak failure: PII still scrubbed, critical error logged, returns success to caller
- Already deleted user: throws `InvalidOperationException`
- Email mismatch: throws `ArgumentException`

    // POST /api/admin/users/{id}/unblock — unblock user

**GET /api/admin/users:**
- Query params: `page`, `pageSize`, `search`, `status`
- Call `_paginatedHandler.HandleAsync(query, ct)`
- Return `Ok(result)`
- No `[Authorize(Roles)]` needed — controller-level attribute covers it

**GET /api/admin/users/{id}:**
- Call `_detailsHandler.HandleAsync(query, ct)`
- Catch `KeyNotFoundException` → return `NotFound(ProblemDetails)`
- Return `Ok(dto)`

**PUT /api/admin/users/{id}:**
- Build `UpdateUserCommand` from body + path `id` + audit context (`AdminSub`, `AdminEmail` from JWT claims)
- Validate with `_updateValidator`
- Catch `ArgumentException` (duplicate email) → `Conflict(ProblemDetails)`
- Catch `KeyNotFoundException` → `NotFound(ProblemDetails)`
- Return `NoContent()` on success

**POST /api/admin/users/{id}/block:**
- Build `BlockUserCommand` with audit context
- Catch `KeyNotFoundException` → `NotFound`
- Catch `InvalidOperationException` (Keycloak unavailable) → `StatusCode(503, ProblemDetails)`
- Return `NoContent()` on success

**POST /api/admin/users/{id}/unblock:** Same pattern as block.

**DELETE /api/admin/users/{id}:**
- Body: `{ "confirmEmail": "user@example.com" }`
- Build `DeleteUserCommand` with audit context
- Catch `ArgumentException` (email mismatch) → `BadRequest(ProblemDetails)`
- Catch `InvalidOperationException` (already deleted) → `Conflict(ProblemDetails)`
- Return `NoContent()` on success

**Audit context extraction (private method):**
```csharp
private (string Sub, string Email) GetAuditContext()
{
    var sub = User.FindFirst("sub")?.Value
        ?? throw new InvalidOperationException("Missing 'sub' claim.");
    var email = User.FindFirst("email")?.Value
        ?? throw new InvalidOperationException("Missing 'email' claim.");
    return (sub, email);
}
```

- All 6 endpoints compile with correct HTTP methods and routes
- `[Authorize(Roles = "admin")]` on controller class
- All endpoints return appropriate status codes (200, 204, 400, 404, 409, 503)
- `ProblemDetails` returned for all error cases (never raw exception messages)
- `GetAuditContext()` extracts `sub` and `email` from JWT claims
